1.df = spark.read.csv("Missing_records.csv",header=True,inferSchema=True)

# Drop any row that contains missing data
2.df.na.drop().show()

# Has to have at least 2 NON-null values
3.df.na.drop(thresh=2).show()

#droping the rows where sales contain null values
4.df.na.drop(subset=["Sales",]).show()


#drop the rows that contains null values
5.df.na.drop(how="any").show()

#drop the rows where all the columns of dataframe contain null values.
6.df.na.drop(how="all").show()                         df.na.drop("all").show()

7. Using agg() for NULL Counts in all columns
from pyspark.sql.functions import col, sum

df.agg(
    sum(col("ID").isNull().cast("int")).alias("ID_nulls"),
    sum(col("Name").isNull().cast("int")).alias("Name_nulls"),
    sum(col("Sales").isNull().cast("int")).alias("Sales_nulls")
).show()

---Fill the missing values

Usually you should specify what columns you want to fill with the subset parameter

7.df.na.fill('No Name',subset=['Name']).show()                 df.na.fill({"Name": "Unknown"}).show()

#fill the data in sales column
8.df.na.fill(0,subset=['Sales']).show()                        df.na.fill({"Sales": 0}).show()


9. df.na.fill({"Name": "Unknown", "Sales": 0}).show()

10. Replace specific values ("" with Unknown)
df.na.replace("", "Unknown").show()
